    public ComponentViewModel Rent()
    {

            return item;

## 🎯 **まとめ：Clean Architectureの利点**

### **1. 保守性**
- **責任の分離**: 各レイヤーが明確な責任を持つ
- **変更の局所化**: UI変更がビジネスロジックに影響しない
- **テストの容易性**: モックによる単体テスト可能

### **2. 拡張性**
- **新機能追加**: 既存コードを変更せずに機能追加
- **プラットフォーム対応**: UIレイヤーのみ変更でマルチプラットフォーム対応
- **外部サービス連携**: インターフェース追加で新サービス対応

### **3. 再利用性**
- **ドメインロジック**: 他のUIフレームワークでも利用可能
- **アプリケーションサービス**: WebAPI化も容易
- **インフラコンポーネント**: 他プロジェクトでも活用可能

### **4. SELLCTプロジェクトでの具体的メリット**
- **メタゲーム機能の分離**: OS統合機能がビジネスロジックと分離
- **セキュリティ境界**: 危険な操作を Infrastructure レイヤーに隔離
- **テスト環境**: 実際のファイル操作なしでゲームロジックテスト可能
- **段階的実装**: レイヤー毎の段階的開発が可能

この Clean Architecture 設計により、SELLCTプロジェクトは保守性・拡張性・テスタビリティを確保しながら、複雑なメタゲーム機能を安全に実装できます。

// ExplorerControlService.cs
public class ExplorerControlService : IExplorerControlService
{
    private readonly IProcessManager _processManager;
    private readonly IRegistryService _registryService;
    private readonly IBackupService _backupService;
    
    public async Task<bool> TerminateExplorerAsync()
    {
        // 1. バックアップ作成
        await _backupService.CreateSystemBackupAsync();
        
        // 2. エクスプローラープロセス特定
        var explorerProcesses = await _processManager
            .GetProcessesByNameAsync("explorer");
            
        // 3. 安全確認
        if (!await ValidateExplorerTerminationSafetyAsync())
            throw new UnsafeOperationException("Explorer termination not safe");
        
        // 4. プロセス終了実行
        foreach (var process in explorerProcesses)
        {
            await _processManager.TerminateProcessAsync(process.Id);
        }
        
        // 5. 復旧情報登録
        await _registryService.RegisterRecoveryInfoAsync();
        
        return !await IsExplorerRunningAsync();
    }
    
    public async Task<bool> RestoreExplorerAsync()
    {
        var startInfo = new ProcessStartInfo
        {
            FileName = "explorer.exe",
            UseShellExecute = true
        };
        
        var process = await _processManager.StartProcessAsync(startInfo);
        await Task.Delay(3000); // 起動待機
        
        return await IsExplorerRunningAsync();
    }
}

public class CommandPromptService : ICommandPromptService
{
    private readonly List<Process> _cmdProcesses = new();
    private readonly IProcessManager _processManager;
    
    public async Task<ICmdInstance> CreateCommandPromptAsync(string command)
    {
        var startInfo = new ProcessStartInfo
        {
            FileName = "cmd.exe",
            Arguments = $"/c {command}",
            UseShellExecute = false,
            CreateNoWindow = false,
            WindowStyle = ProcessWindowStyle.Normal
        };
        
        var process = await _processManager.StartProcessAsync(startInfo);
        _cmdProcesses.Add(process);
        
        return new CmdInstance(process, this);
    }
    
    public async Task TerminateAllCommandPromptsAsync()
    {
        foreach (var process in _cmdProcesses.ToList())
        {
            try
            {
                await _processManager.TerminateProcessAsync(process.Id);
                _cmdProcesses.Remove(process);
            }
            catch (Exception ex)
            {
                // ログ記録のみ、処理継続
                LogError($"Failed to terminate CMD {process.Id}: {ex.Message}");
            }
        }
    }
}

        // 2. レジストリエントリクリーンアップ
        await CleanupRegistryEntriesAsync();

// SafetyValidatorService.cs
public class SafetyValidatorService : ISafetyValidatorService
{
    public async Task<bool> ValidateOperationSafetyAsync(SystemOperation operation)
    {
        switch (operation.Type)
        {
            case OperationType.TerminateExplorer:
                return await ValidateExplorerTerminationAsync();
                
            case OperationType.ModifyRegistry:
                return await ValidateRegistryModificationAsync(operation);
                
            case OperationType.CreateStartupEntry:
                return await ValidateStartupEntryAsync(operation);
                
            default:
                return false;
        }
    }
    
    private async Task<bool> ValidateExplorerTerminationAsync()
    {
        // 1. 管理者権限チェック
        if (!IsRunningAsAdministrator())
            return false;
            
        // 2. ウイルス対策ソフト確認
        if (await IsAntivirusActiveAsync())
            return false;
            
        // 3. 重要プロセス実行中チェック
        if (await AreCriticalProcessesRunningAsync())
            return false;
            
        return true;
    }
}

// NavigationService.cs
public class NavigationService : INavigationService
{
    private readonly Stack<IViewModel> _navigationStack = new();
    private Window _currentWindow;
    
    public async Task NavigateToAsync<T>() where T : IViewModel
    {
        var viewModel = ServiceLocator.GetService<T>();
        var window = CreateWindowForViewModel(viewModel);
        
        _navigationStack.Push(viewModel);
        
        if (_currentWindow != null)
        {
            _currentWindow.Hide();
        }
        
        _currentWindow = window;
        window.Show();
        
        await viewModel.OnNavigatedToAsync();
    }
    
    public async Task NavigateBackAsync()
    {
        if (_navigationStack.Count <= 1) return;
        
        var currentViewModel = _navigationStack.Pop();
        await currentViewModel.OnNavigatedFromAsync();
        
        var previousViewModel = _navigationStack.Peek();
        var window = CreateWindowForViewModel(previousViewModel);
        
        _currentWindow?.Close();
        _currentWindow = window;
        window.Show();
        
        await previousViewModel.OnNavigatedBackToAsync();
    }
}

// AnimationService.cs
public class AnimationService : IAnimationService
{
    public async Task FadeInAsync(UIElement element, TimeSpan duration)
    {
        var animation = new DoubleAnimation
        {
            From = 0,
            To = 1,
            Duration = duration,
            EasingFunction = new QuadraticEase()
        };
        
        element.BeginAnimation(UIElement.OpacityProperty, animation);
        await Task.Delay(duration);
    }
    
    public async Task ShakeAsync(UIElement element, int intensity = 5)
    {
        var transform = new TranslateTransform();
        element.RenderTransform = transform;
        
        for (int i = 0; i < 10; i++)
        {
            var offsetX = (i % 2 == 0) ? intensity : -intensity;
            
            var animation = new DoubleAnimation
            {
                To = offsetX,
                Duration = TimeSpan.FromMilliseconds(50)
            };
            
            transform.BeginAnimation(TranslateTransform.XProperty, animation);
            await Task.Delay(50);
        }
        
        // 元の位置に戻す
        var resetAnimation = new DoubleAnimation
        {
            To = 0,
            Duration = TimeSpan.FromMilliseconds(100)
        };
        transform.BeginAnimation(TranslateTransform.XProperty, resetAnimation);
    }
}
